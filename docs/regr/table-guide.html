数据准备与复现

使用 RegR 自带的 SEER 甲状腺髓样癌队列。保留完整模型变量、正随访时间及肿瘤大小在 0–200 mm 内的记录，再固定种子抽取最多 800 人。stage 合并为 T1/T2 与其余分期，surgery 为无手术与任一手术。它们是本页的演示编码。时间单位沿用原始队列的月。只发布汇总结果。

library(RegR)
library(ggplot2)
library(survival)
data("seer_thyroid_mtc_2026", package = "RegR")
d <- as.data.frame(seer_thyroid_mtc_2026)
d[["DSS"]] <- as.integer(as.character(d[["DSS"]]))
d[["OS"]] <- as.integer(as.character(d[["OS"]]))
needed <- c("time", "DSS", "OS", "Age", "Tum", "Sex", "Race", "T_stage", "Sur")
d <- d[complete.cases(d[needed]) & d[["time"]] > 0 &
       d[["Tum"]] > 0 & d[["Tum"]] < 200, , drop = FALSE]
d[["stage"]] <- factor(ifelse(d[["T_stage"]] %in% c("T1", "T2"), "Early", "Other"))
d[["surgery"]] <- factor(ifelse(d[["Sur"]] == "No surgery", "No", "Yes"),
                          levels = c("No", "Yes"))
set.seed(20261001)
d <- droplevels(d[sample.int(nrow(d), min(800L, nrow(d))), , drop = FALSE])

共用计算与返回对象

表格演示另建 td，将 DSS 转为分类变量；生存模型仍使用上面数值 0/1 的 d。output_format=“none” 保留 gtsummary 对象，“gt” 返回 gt 表，默认 “flextable” 适合 Word。不要把表格对象类型和统计方法混为一谈。

接口 用途 当前输出

当前名称字典与森林图一致

2026-10-20 按已推送 RegR a2f2553 核对，get_tb_gtsummary() 合并 colname_map 的 name_map_seer 也用于 plt_eff()、plt_eff2() 和 plt_eff_pair() 的默认行名。字典补入 NACT="Neoadjuvant chemotherapy"；Age、Grade_3、分期等已有表格标签现在也由森林图使用，减少同一变量在图和表中的不同叫法。

用户映射仅覆盖给出的键，未列出的变量继续用字典；标签只是显示层，不修改原始列名、因子水平或参考组。需要原名时将该键映射到自身。森林图的可运行多模型例子展示了NACT、Age与Grade_3的实际输出；本页既有表格保留其原执行记录。

精度与自动选列（2026-10-24）

2026-10-24 按已推送 RegR 2c01bb0 在 Windows / WSL R 4.4.3 执行。get_tb_gtsummary() 的连续摘要默认改为 0 位小数，支持按原始列名分别指定；下方历史表保留原输出，代码显式设 2 位以复现旧精度。get_gt_cat() 仍自行使用 2 位，不随此默认一起改变。

默认值、变量名与显示标签

library(RegR)
d_digits <- data.frame(
  group = factor(rep(c("a", "b"), each = 4)),
  age = c(21.32, 24.18, 30.77, 33.29, 22.64, 25.43, 29.91, 35.12),
  time = c(10, 12, 14, 16, 20, 22, 24, 26),
  sex = factor(rep(c("f", "m"), 4)))
make_digits <- function(...) get_tb_gtsummary(d_digits, group_var = "group",
  com_var = c("age", "time", "sex"), colname_map = c(age = "Age at diagnosis"),
  output_format = "none", ...)
default_digits <- make_digits()
custom_digits <- make_digits(digits_continuous = c(.default = 2, time = 0))
one_digit <- make_digits(digits_continuous = c(age = 1))
overall_digits <- function(x, v) {
  b <- x[["table_body"]]; b[["stat_0"]][b[["variable"]] == v & b[["row_type"]] == "label"]
}
reject_digits <- function(x) inherits(tryCatch(make_digits(digits_continuous = x), error = identity), "error")
gt::gtsave(gtsummary::as_gt(custom_digits),
  file.path(tempdir(), "20261024-update-table-digits.html"))

模拟Fisher脚注与Word制表符（2026-11-03）

分层与期刊风格

strata_var 为每个层生成相应的基线表；它不等同于回归模型调整。journal_style 会改变 gtsummary 的全局主题，后续调用前需重置，避免参数比较混入前一张表的格式。

UMS原始统计量与舍入边界（2026-11-03）

model_stats现在直接从保留的模型提取未舍入HR/OR、Wald区间和P值，显著性符号也用该P值计算。不要解析显示字符串的0.05来重建星号，或把p<.001当成P等于0。下面用120行合成数据数值求根构造临界P值，分别核对Cox和Logistic；这是显示边界示例，不是临床结果。

library(RegR)
precision_rows <- local({
  set.seed(41)
  cox_data <- data.frame(time = rexp(120), DSS = 1L)
  cox_noise <- rnorm(120)
  cox_score <- -log(cox_data[["time"]])
  cox_fit <- function(a) {
    d <- cox_data
    d[["x"]] <- cox_noise + a * cox_score
    survival::coxph(survival::Surv(time, DSS) ~ x, data = d)
  }
  cox_center <- uniroot(function(a) summary(cox_fit(a))[["coefficients"]][1, "z"],
    c(-3, 3), tol = 1e-12)[["root"]]
  set.seed(5)
  y <- rep(0:1, each = 60)
  w <- rnorm(120)
  log_center <- mean(w[y == 0]) - mean(w[y == 1])
  log_fit <- function(a) glm(y ~ x, data = data.frame(y = y, x = w + a * y), family = binomial())
  dplyr::bind_rows(lapply(c("Cox", "Logistic"), function(model) {
    fit_at <- if (model == "Cox") cox_fit else log_fit
    center <- if (model == "Cox") cox_center else log_center
    p_at <- function(a) summary(fit_at(a))[["coefficients"]][if (model == "Cox") 1 else 2, "Pr(>|z|)"]
    dplyr::bind_rows(lapply(c(.050004, .010004, .001004, .0004), function(target) {
      shift <- uniroot(function(a) p_at(a) - target, c(center, center + 3), tol = 1e-12)[["root"]]
      d <- if (model == "Cox") cox_data else data.frame(y = y)
      d[["x"]] <- if (model == "Cox") cox_noise + shift * cox_score else w + shift * y
      result <- run_quiet(get_tb_ums(d, "x", surv = if (model == "Cox") TRUE else "y", threshold = 1))
      stats <- attr(result, "model_stats")
      stats <- stats[stats[["id"]] == "x", ]
      metric <- if (model == "Cox") "HR" else "OR"
      star <- result[[paste0(metric, " (univariable)_p")]]
      reference <- summary(fit_at(shift))[["coefficients"]]
      beta <- reference[if (model == "Cox") 1 else 2, if (model == "Cox") "coef" else "Estimate"]
      stopifnot(all(abs(stats[["p"]] - p_at(shift)) < 1e-12),
        all(abs(stats[[metric]] - exp(beta)) < 1e-12),
        identical(star, UtilsR::stat_pval(p_at(shift), mode = "stars")))
      data.frame(Model = model, Target = target, P_unrounded = stats[["p"]][1],
        Display = stats[["stats"]][1], Stars = star)
    }))
  }))
})
gt::gtsave(gt::fmt_number(gt::gt(precision_rows), columns = c(Target, P_unrounded), decimals = 8),

UMS接受可无损转成数值的integer64、Date/POSIX时间、difftime和Duration：Date按天，POSIX/Duration按秒，difftime沿用记录单位；日历Period应先明确换算单位。SPSS用户缺失按NA处理。列表、矩阵、复数、精度会损失的integer64及显式NA因子水平需要调用方先处理。

这里的autoReg路径要求能对应水平效应的命名treatment编码；有序、sum或其他无法匹配水平的编码会拒绝。它与get_um的编码恢复不同。本节保留 RegR 769f460c 的历史错误示例。当前 7dad336b 已为恢复后重复的标识添加数字后缀，原变量身份继续保留，见下方标识对照。

本节实际执行了SPSS缺失对照、非法列、重名和sum编码错误；日期/时长单位、有序因子和其他编码类型按固定源码核对。

UMS 缺失策略、空队列和批量输出（2026-11-06）

使用边界、保存与复现记录

本页实际运行并展示了 9 张汇总表，未发布患者记录。不同格式不保证数值位数完全相同，比较时应看未格式化数据而不是字符串。期刊主题、列名映射和首字母处理属于显示行为；建模结局、分组、缺失处理和百分比分母必须另行明确。get_tb_autoreg 内部对缺失值采用 Unknown 路线；混合数值/缺失数据要先核对类型，本页使用预先筛选的完整记录。
