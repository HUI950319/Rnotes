函数选择与重要参数

函数 重要参数 当前行为
lv() ...、pattern=NULL tidyselect 选择列或正则选择；教程显式给 data，避免依赖全局 data/data_all
lv() group=NULL、count=NULL group 是分组摘要；count 是一个或多个交叉计数变量，优先走交叉表模式
lv() cat_args=list() 非空时额外调用 RegR::get_gt_cat；依赖 RegR，结果并不替代基础检视表
check_na() .../pattern、special=NULL、show_all=FALSE 返回 gt 报告表；默认只列问题变量，show_all 包括无缺失变量；pattern 优先于列选择，两者同时给出时警告；special 替换内置特殊值词表
plt_na() output="both" both/both_reverse/matrix/percentage 四种布局
plt_na() sort="desc"、name.map=NULL 按缺失率排序；命名字符向量替换显示名称
plt_na() miss_palette 前三色依次为存在、缺失和 30% 参考线；只显示存在 NA 的列
plt_upset() vars、levels=NA、output、label 把每列缺失行当作一个集合，查看联合缺失
impute_na_knn() k=5、verbose=TRUE k 为邻居数，不超过每个待插补变量的观察供体数；数字用距离加权均值
check_size() ...、pattern=NULL、envir=.GlobalEnv、method 检查指定环境自身的绑定；默认 object.size，可选 lobstr；返回 gt，数值大小仍为 MB
check_system() return_result=FALSE、show_warnings=FALSE 返回系统与 R 环境 list；WSL 结果描述 WSL 进程所在环境

# 本页导出 helper：实际编织调用，保留其原生 gt HTML；不改包或 lv 返回值。
knit_lv <- function(input, call) {
  local({
    old_to <- knitr::opts_knit$get("rmarkdown.pandoc.to")
    on.exit(knitr::opts_knit$set(rmarkdown.pandoc.to = old_to))
    knitr::opts_knit$set(rmarkdown.pandoc.to = "html")
    e <- new.env(parent = environment()); e[["dat_lv"]] <- input
    fence_lv <- strrep("`", 3)
    body_lv <- knitr::knit(text = c(
      paste0(fence_lv, "{r, results='asis', echo=FALSE, message=FALSE, error=FALSE}"),
      paste0("ans_lv <- withVisible(", call, ")"),
      "stopifnot(identical(ans_lv[['value']], dat_lv), !ans_lv[['visible']])",
      fence_lv), envir = e, quiet = TRUE)
    stopifnot(!grepl("## <div", body_lv, fixed = TRUE), !grepl("# A tibble", body_lv, fixed = TRUE))
    xml2::read_html(paste0("<html><head><meta charset='utf-8'></head><body>", body_lv, "</body></html>"))
  })
}
lv_tables <- function(x) xml2::xml_find_all(x, "//table[contains(@class,'gt_table')]")
save_lv <- function(doc, i, name) {
  # gt 的父 div 连同样式一起保存，统计内容和格式均来自原生输出。
  fragment <- as.character(xml2::xml_parent(lv_tables(doc)[[i]]))
  writeLines(paste0("<!doctype html><html><head><meta charset='utf-8'></head><body>",
    fragment, "</body></html>"), file.path(tempdir(), name), useBytes = TRUE)
}
